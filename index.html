offline-form.js

Auto-save any HTML form. Restore on reload. Retry on reconnect.
2 lines of code. Zero dependency.

📦 npm v0.2.2 ⭐ GitHub 📖 Docs
npm install offline-form
💾

Auto-save

Every keystroke saved automatically with debounce

🔄

Auto-restore

Data restored silently on page reload or crash

📶

Offline retry

Submit queued and retried automatically on reconnect

⚡ Quick start

// Install // npm install offline-form import OfflineForm from 'offline-form' // That's it 👇 OfflineForm.watch('#my-form') // With options OfflineForm.watch('#my-form', { debounce: 800, autoSubmit: true, toast: true, onRestore: (data) => {}, onSave: (data) => {}, })

📋 All options

OfflineForm.watch(selector, { debounce: 800, // ms key: 'my-key', // storage key toast: true, // restore toast autoSubmit: false, // retry on reconnect onSave: (data) => {}, onRestore: (data) => {}, onRetry: (item) => {}, }) // Methods OfflineForm.unwatch('#form') OfflineForm.clear('#form') OfflineForm.getAll()

⚛️ React

import OfflineForm from 'offline-form' import { useEffect } from 'react' function ContactForm() { useEffect(() => { OfflineForm.watch('#contact') return () => OfflineForm.unwatch('#contact') }, []) return <form id="contact">...</form> }
